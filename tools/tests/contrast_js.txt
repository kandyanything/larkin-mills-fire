/* Lists visible text whose contrast with its background is below WCAG AA (4.5, or 3 for large text). Used by nwk1_b.py. */
(() => {
  function rgba(s) { var m = s.match(/rgba?\(([^)]+)\)/); if (!m) return null; var p = m[1].split(',').map(parseFloat); return [p[0], p[1], p[2], p.length > 3 ? p[3] : 1]; }
  function lum(c) { var a = c.slice(0, 3).map(function (v) { v /= 255; return v <= .03928 ? v / 12.92 : Math.pow((v + .055) / 1.055, 2.4); }); return .2126 * a[0] + .7152 * a[1] + .0722 * a[2]; }
  function bg(el) { var stack = []; for (var e = el; e && e.nodeType === 1; e = e.parentElement) { var cs = getComputedStyle(e); if (cs.backgroundImage !== 'none' && !/gradient/.test(cs.backgroundImage)) return null; var c = rgba(cs.backgroundColor); if (c && c[3] > 0) { stack.push(c); if (c[3] >= .99) break; } } var out = [10, 18, 36]; for (var i = stack.length - 1; i >= 0; i--) { var c = stack[i]; out = [c[0] * c[3] + out[0] * (1 - c[3]), c[1] * c[3] + out[1] * (1 - c[3]), c[2] * c[3] + out[2] * (1 - c[3])]; } return out; }
  var bad = [], seen = new Set();
  document.querySelectorAll('body *').forEach(function (el) {
    if (!el.offsetParent && getComputedStyle(el).position !== 'fixed') return;
    var own = Array.prototype.some.call(el.childNodes, function (n) { return n.nodeType === 3 && n.textContent.trim().length > 1; });
    if (!own) return;
    if (el.closest('dialog:not([open]), [hidden], .wl, svg, .hero, .tribute, .ad, .partner-banner, .pb, .bdoor, .sheet-photo, figure')) return;
    var cs = getComputedStyle(el); if (cs.visibility === 'hidden' || +cs.opacity < .5) return;
    var b = bg(el); if (!b) return; var f = rgba(cs.color); if (!f) return;
    var L1 = lum(f), L2 = lum(b), ratio = (Math.max(L1, L2) + .05) / (Math.min(L1, L2) + .05);
    var big = parseFloat(cs.fontSize) >= 24 || (parseFloat(cs.fontSize) >= 18.6 && +cs.fontWeight >= 700);
    if (ratio < (big ? 3 : 4.5)) { var key = (el.className || el.tagName) + '|' + el.textContent.trim().slice(0, 30); if (!seen.has(key)) { seen.add(key); bad.push([ratio.toFixed(2), String(el.className || el.tagName).slice(0, 40), el.textContent.trim().slice(0, 40), cs.color, 'bg rgb(' + b.map(Math.round).join(',') + ')']); } }
  });
  return bad;
})()
